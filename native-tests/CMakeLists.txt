cmake_minimum_required(VERSION 3.16)
project(render_file_format_tests LANGUAGES CXX)
enable_testing()
add_library(blend_probe STATIC ../app/src/main/cpp/blend_probe.cpp)
target_include_directories(blend_probe PUBLIC ../app/src/main/cpp)
target_compile_features(blend_probe PUBLIC cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(blend_probe PRIVATE -Wall -Wextra -Werror)
endif()
add_executable(blend_probe_tests blend_probe_tests.cpp)
target_link_libraries(blend_probe_tests PRIVATE blend_probe)
add_test(NAME blend_probe_tests COMMAND blend_probe_tests)
